% ─── Slide 9 — Decisões de design ───────────────────────────
\begin{frame}{Decisões de design}
  \kicker{TRADE-OFFS EXPLÍCITOS}
  \vspace{0.2cm}

  \footnotesize
  \begin{itemize}\itemsep0.35em
    \item \textbf{\textcolor{accent}{DA-01}} \(\cdot\) Frontend e backend em processos separados\\
          {\scriptsize\textcolor{textmuted}{Permite que o processamento Python pesado não bloqueie a UI}}
    \item \textbf{\textcolor{accent}{DA-02}} \(\cdot\) Imagens via base64 dentro do JSON\\
          {\scriptsize\textcolor{textmuted}{API simples, uma resposta = todos os dados; trade-off: +33\% de tamanho}}
    \item \textbf{\textcolor{accent}{DA-04}} \(\cdot\) Reprojeção dinâmica (WGS84 \(\rightarrow\) UTM)\\
          {\scriptsize\textcolor{textmuted}{Frontend trabalha em lat/lon, backend trata projeções via pyproj}}
    \item \textbf{\textcolor{accent}{DA-06}} \(\cdot\) SentinelProcessor isolado das rotas HTTP\\
          {\scriptsize\textcolor{textmuted}{Permite testes unitários sem subir o servidor}}
    \item \textbf{\textcolor{accent}{DA-07}} \(\cdot\) Tipagem end-to-end (TS + Pydantic)\\
          {\scriptsize\textcolor{textmuted}{Erros de contrato detectados em build, não em produção}}
    \item \textbf{\textcolor{accent}{DA-09}} \(\cdot\) Suavização gaussiana com máscara\\
          {\scriptsize\textcolor{textmuted}{Evita ``sangramento'' de valores nas bordas do polígono}}
  \end{itemize}
\end{frame}

% ─── Slide 10 — Fluxo NDVI ──────────────────────────────────
\begin{frame}{Fluxo de cálculo de NDVI}
  \kicker{DIAGRAMA DE SEQUÊNCIA}
  \vspace{0.2cm}

  \footnotesize
  \begin{enumerate}\itemsep0.18em
    \item Usuário seleciona talhão e índice ``NDVI'' na UI
    \item Frontend envia \texttt{POST /calculate-index} com coordenadas WGS84
    \item Backend resolve produto Sentinel-2 .SAFE mais recente
    \item rasterio lê bandas B08 (NIR) e B04 (Red) em 10\,m
    \item pyproj reprojeta polígono WGS84 \(\rightarrow\) CRS nativo da imagem (UTM)
    \item rasterio.mask recorta as bandas ao polígono
    \item NumPy calcula NDVI = (NIR \(-\) Red) / (NIR + Red), clipa a \([-1, 1]\)
    \item Pillow gera PNG colorido (gradiente \textcolor{ndvilow}{marrom} \(\rightarrow\) \textcolor{ndvimid}{amarelo} \(\rightarrow\) \textcolor{ndvihigh}{verde})
    \item Backend retorna JSON com base64, estatísticas, histograma e dados 3D
    \item Frontend sobrepõe imagem ao mapa Leaflet + atualiza painel de stats
  \end{enumerate}
\end{frame}

% ─── Slide 14 — Limitações ──────────────────────────────────
\begin{frame}{Limitações conhecidas}
  \kicker{O QUE NÃO ESTÁ NO MVP}
  \vspace{0.3cm}

  \footnotesize
  \begin{itemize}\itemsep0.4em
    \item \textbf{\textcolor{accent}{Sem persistência de dados}} \(\cdot\) talhões desenhados são perdidos ao recarregar
    \item \textbf{\textcolor{accent}{Classificação supervisionada simulada}} \(\cdot\) não usa modelo treinado real
    \item \textbf{\textcolor{accent}{Imagens Sentinel-2 baixadas manualmente}} \(\cdot\) sem download automático via API Copernicus
    \item \textbf{\textcolor{accent}{Single-user em localhost}} \(\cdot\) sem autenticação ou multi-tenancy
    \item \textbf{\textcolor{accent}{Cobertura de I/O limitada}} \(\cdot\) read\_band/crop dependem de .SAFE real, testes manuais cobrem
  \end{itemize}
\end{frame}
